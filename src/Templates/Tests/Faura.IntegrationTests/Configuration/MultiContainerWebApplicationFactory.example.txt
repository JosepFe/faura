/// <summary>
/// Example: CustomWebApplicationFactory with multiple containers.
/// Shows how to configure PostgreSQL + MongoDB + Redis for integration tests.
/// 
/// Usage:
/// 1. Copy this class and rename it to your needs
/// 2. Update appsettings.Test.json with container configurations
/// 3. Implement ConfigureTestServices and ConfigureTestDatabase
/// </summary>
public class MultiContainerWebApplicationFactory : BaseWebApplicationFactory<Program>
{
    private TestContainerInstance<PostgresContainerConfiguration>? _postgresContainer;
    private TestContainerInstance<MongoContainerConfiguration>? _mongoContainer;
    private TestContainerInstance<RedisContainerConfiguration>? _redisContainer;

    protected override async Task<IConfiguration> ConfigureTestContainersAsync(
        IConfiguration configuration
    )
    {
        var containerOptions = configuration.GetSection("Containers").Get<TestContainerOptions>();
        ArgumentNullException.ThrowIfNull(containerOptions);

        // Start PostgreSQL Container
        if (containerOptions.Postgres != null)
        {
            var pgConfig = new PostgresContainerConfiguration(containerOptions.Postgres);
            _postgresContainer = new TestContainerInstance<PostgresContainerConfiguration>(pgConfig);
            await _postgresContainer.StartAsync();
        }

        // Start MongoDB Container
        if (containerOptions.Mongo != null)
        {
            var mongoConfig = new MongoContainerConfiguration(containerOptions.Mongo);
            _mongoContainer = new TestContainerInstance<MongoContainerConfiguration>(mongoConfig);
            await _mongoContainer.StartAsync();
        }

        // Start Redis Container
        if (containerOptions.Redis != null)
        {
            var redisConfig = new RedisContainerConfiguration(containerOptions.Redis);
            _redisContainer = new TestContainerInstance<RedisContainerConfiguration>(redisConfig);
            await _redisContainer.StartAsync();
        }

        // Override configuration with container connection strings
        var configOverrides = new Dictionary<string, string?>();

        if (_postgresContainer != null)
        {
            configOverrides["ConnectionStrings:DefaultConnection"] = _postgresContainer.ConnectionString;
            configOverrides["ConnectionStrings:Employee"] = _postgresContainer.ConnectionString;
        }

        if (_mongoContainer != null)
        {
            configOverrides["MongoDb:ConnectionString"] = _mongoContainer.ConnectionString;
            configOverrides["MongoDb:DatabaseName"] = "test_db";
        }

        if (_redisContainer != null)
        {
            configOverrides["Redis:ConnectionString"] = _redisContainer.ConnectionString;
        }

        return new ConfigurationBuilder()
            .AddConfiguration(configuration)
            .AddInMemoryCollection(configOverrides)
            .Build();
    }

    /// <summary>
    /// Clean up containers when factory is disposed.
    /// </summary>
    public new async ValueTask DisposeAsync()
    {
        if (_postgresContainer != null)
            await _postgresContainer.DisposeAsync();
        
        if (_mongoContainer != null)
            await _mongoContainer.DisposeAsync();
        
        if (_redisContainer != null)
            await _redisContainer.DisposeAsync();

        await base.DisposeAsync();
    }
}
